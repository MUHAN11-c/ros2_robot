复现环境

# Jazzy + turtlesim(源码编译) + rosidl 中文路径补丁，见 /tmp/ros_env.sh 同款配置
source /opt/ros/jazzy/setup.bash
cd code/<工作空间>
colcon build
source install/setup.bash
# 然后按教程各节命令运行
